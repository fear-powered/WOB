const pages = dv.pages('"Особое/Магические предметы"')
    .where(p => p.file.path !== dv.current().file.path)
    .array();

const types = [...new Set(pages.map(p => p.тип).filter(Boolean))];
const rarities = [...new Set(pages.map(p => p.редкость).filter(Boolean))];

const root = dv.container;

const controls = root.createDiv();
controls.style.display = "flex";
controls.style.flexWrap = "wrap";
controls.style.gap = "10px";
controls.style.marginBottom = "12px";

function createFilter(label, values) {
    const wrapper = controls.createDiv();
    wrapper.createEl("label", {text: label});

    const select = wrapper.createEl("select");
    select.style.display = "block";
    select.style.padding = "5px";

    select.createEl("option", {text: "Все", value: ""});

    values.forEach(value => {
        select.createEl("option", {
            text: String(value),
            value: String(value)
        });
    });

    return select;
}

const typeFilter = createFilter("Тип", types);
const rarityFilter = createFilter("Редкость", rarities);
const attunementFilter = createFilter("Настройка", ["Да", "Нет"]);

const tableContainer = root.createDiv();

function renderTable() {
    const filtered = pages.filter(p => {
        const typeMatch = !typeFilter.value ||
            String(p.тип) === typeFilter.value;

        const rarityMatch = !rarityFilter.value ||
            String(p.редкость) === rarityFilter.value;

        const attunementMatch =
            !attunementFilter.value ||
            (attunementFilter.value === "Да"
                ? p.настройка === true
                : p.настройка === false);

        return typeMatch && rarityMatch && attunementMatch;
    });

    // Очищаем только старую таблицу
    tableContainer.replaceChildren();

    // Создаём новую таблицу в том же контейнере
    const table = tableContainer.createEl("table");
    const thead = table.createEl("thead");
    const headerRow = thead.createEl("tr");

    const headers = [
        `Предмет (${filtered.length})`,
        "Тип",
        "Редкость",
        "Настройка"
    ];

    headers.forEach(header => {
        headerRow.createEl("th", {text: header});
    });

    const tbody = table.createEl("tbody");

    filtered.forEach(p => {
        const row = tbody.createEl("tr");

        const nameCell = row.createEl("td");
        const link = nameCell.createEl("a", {
            text: p.file.name,
            cls: "internal-link"
        });

        link.setAttribute("href", p.file.path);
        link.setAttribute("data-href", p.file.path);

        row.createEl("td", {text: String(p.тип ?? "—")});
        row.createEl("td", {text: String(p.редкость ?? "—")});

        row.createEl("td", {
            text: p.настройка === true ? "Да" :
                  p.настройка === false ? "Нет" : "—"
        });
    });
}

// Обновление при изменении фильтров
[typeFilter, rarityFilter, attunementFilter].forEach(select => {
    select.addEventListener("change", renderTable);
});

renderTable();
